import React, { useState, useEffect } from 'react';
import axios from 'axios';

const API_URL = 'http://localhost:4000/api';

function App() {
  const [token, setToken] = useState(localStorage.getItem('token') || '');
  const [role, setRole] = useState(localStorage.getItem('role') || '');
  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [stores, setStores] = useState([]);
  const [ratingInputs, setRatingInputs] = useState({});

  const login = async () => {
    try {
      const res = await axios.post(`${API_URL}/login`, { email, password });
      setToken(res.data.token);
      setRole(res.data.role);
      localStorage.setItem('token', res.data.token);
      localStorage.setItem('role', res.data.role);
      alert('Logged in');
      fetchStores(res.data.token);
    } catch {
      alert('Login failed');
    }
  };

  const logout = () => {
    setToken('');
    setRole('');
    localStorage.removeItem('token');
    localStorage.removeItem('role');
    setStores([]);
  };

  const fetchStores = async (authToken = token) => {
    try {
      const res = await axios.get(`${API_URL}/stores`, {
        headers: { Authorization: `Bearer ${authToken}` }
      });
      setStores(res.data);
    } catch {
      alert('Failed to fetch stores');
    }
  };

  const submitRating = async (storeId) => {
    const rating = ratingInputs[storeId];
    if (!rating || rating < 1 || rating > 5) {
      alert('Rating must be 1 to 5');
      return;
    }
    try {
      await axios.post(`${API_URL}/stores/${storeId}/rate`, { rating }, {
        headers: { Authorization: `Bearer ${token}` }
      });
      alert('Rating submitted');
      fetchStores();
    } catch {
      alert('Failed to submit rating');
    }
  };

  useEffect(() => {
    if (token) fetchStores();
  }, [token]);

  if (!token) {
    return (
      <div style={{ padding: 20 }}>
        <h2>Login</h2>
        <input placeholder="Email" value={email} onChange={e => setEmail(e.target.value)} /><br />
        <input placeholder="Password" type="password" value={password} onChange={e => setPassword(e.target.value)} /><br />
        <button onClick={login}>Login</button>
      </div>
    );
  }

  return (
    <div style={{ padding: 20 }}>
      <h2>Welcome, {role}</h2>
      <button onClick={logout}>Logout</button>

      <h3>Stores</h3>
      <table border="1" cellPadding="5">
        <thead>
          <tr>
            <th>Name</th>
            <th>Address</th>
            <th>Average Rating</th>
            {role === 'normal' && <th>Your Rating</th>}
            {role === 'normal' && <th>Submit/Update Rating</th>}
          </tr>
        </thead>
        <tbody>
          {stores.map(store => (
            <tr key={store.id}>
              <td>{store.name}</td>
              <td>{store.address}</td>
              <td>{store.avg_rating.toFixed(2)}</td>
              {role === 'normal' && <td>{store.user_rating || '-'}</td>}
              {role === 'normal' && (
                <td>
                  <input
                    type="number"
                    min="1"
                    max="5"
                    value={ratingInputs[store.id] || ''}
                    onChange={e => setRatingInputs({ ...ratingInputs, [store.id]: e.target.value })}
                    style={{ width: 50 }}
                  />
                  <button onClick={() => submitRating(store.id)}>Submit</button>
                </td>
              )}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}

export default App;
